cmake_minimum_required(VERSION 3.16)
project(pam_mixed VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(Threads REQUIRED)
add_executable(pam_mixed src/main.cpp src/engine.cpp)
target_include_directories(pam_mixed PRIVATE include)
target_link_libraries(pam_mixed PRIVATE Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(pam_mixed PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(mixed_engine_tests tests/test_engine.cpp src/engine.cpp)
  target_include_directories(mixed_engine_tests PRIVATE include)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(mixed_engine_tests PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME mixed_engine COMMAND mixed_engine_tests)
endif()
find_package(Python3 COMPONENTS Interpreter QUIET)
if(BUILD_TESTING AND Python3_Interpreter_FOUND)
  add_test(NAME mixed_cli COMMAND ${CMAKE_COMMAND} -E env
    PAM_MIXED_BIN=$<TARGET_FILE:pam_mixed>
    ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py)
endif()
